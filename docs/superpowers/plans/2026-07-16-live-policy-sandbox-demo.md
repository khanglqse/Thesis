# Live Policy Sandbox Demo — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A single self-contained web page that runs the real trained `haichau_rush` DQN weights live in the browser on one Hải Châu intersection, making the agent's traffic-light control visible (big signal heads, decision flash, Q-values, live decision log) and quantified (shadow Fixed-Time scoreboard) — with no Docker/CityFlow at demo time.

**Architecture:** A host-Python export script serializes the trained `.npz` weights + the intersection's lane/phase/geometry structure + parity samples into `frontend/sandbox_model.json`. A pure-logic JS module `frontend/sandbox_core.js` (Net + Encoder + MicroModel + controllers) reproduces `SharedDQN.greedy_action` exactly (proven by a Node parity test) and simulates traffic. A rendering module `frontend/sandbox.js` draws the intersection on a Canvas and wires the DOM. Served by the existing `scripts/serve_replay.py`.

**Tech Stack:** Python 3.13 (host) for export; NumPy; vanilla JavaScript + HTML5 Canvas 2D (no build step, no runtime deps); Node.js built-in test runner (`node --test`) and pytest for tests.

## Global Constraints

- **`src/` stays Python-3.6-safe** (no `from __future__ import annotations`). The new `scripts/export_sandbox_model.py` is **host-only (Python ≥3.10)** and only *imports* `src` modules, so modern syntax is fine there.
- **No Docker/CityFlow/container at demo time.** Export runs host Python; the page runs pure browser JS.
- **Faithfulness gate:** the JS forward pass MUST reproduce `SharedDQN` on the exported parity samples within abs tol `1e-3` (float32-vs-float64 headroom) and pick the same greedy action. If parity fails, the demo is not shippable.
- **Headline model constants (verbatim, `haichau_rush`):** `count_norm=5`, `state_mode=basic`, `hidden=128`, `obs_dim=33` (incoming `max_in=14` | outgoing `max_out=13` | green one-hot `max_green=6`), `n_actions=6`, `action_interval=10`, `min_green=10`, `yellow_time=3`, Fixed-Time green `=25`. Weights: `results/haichau_rush/dqn_haichau_rush.npz`. Roadnet: `data/danang/roadnet_haichau_rush.json`.
- **obs layout (verbatim from `ObservationEncoder.encode`, basic mode):** `v[k]=incoming_count/count_norm` for `k<len(incoming)`; `v[max_in+k]=outgoing_count/count_norm`; `v[max_in+max_out+green_index]=1`. Incoming/outgoing counts are **all vehicles per lane** (`lane_vehicle_counts`), not just queued.
- **UI copy in Vietnamese** (match `frontend/demo-presets.js`).
- **Honesty:** on-screen note that this is an illustrative micro-model running the *real trained policy*; quantitative 53-node results live in Chapter 4.

```python
#!/usr/bin/env python3
"""Export a trained SharedDQN + one intersection's structure to JSON for the
browser sandbox demo. Host-only (Python 3.13); imports 3.6-safe src modules.

  PYTHONPATH=src python scripts/export_sandbox_model.py --area haichau_rush
  PYTHONPATH=src python scripts/export_sandbox_model.py --area haichau_rush --list
"""
import argparse
import json
import math
from pathlib import Path

import numpy as np

from traffic_rl.network.roadnet import RoadNet
from traffic_rl.sim.encoder import ObservationEncoder
from traffic_rl.agents.dqn import SharedDQN

ROOT = Path(__file__).resolve().parent.parent

# Per-scenario encoder config (kept in sync with the exp_*.sh training overrides).
SCENARIOS = {
    "haichau_rush":  {"count_norm": 5.0,  "state_mode": "basic", "fixed_green": 25},
    "haichau_calib": {"count_norm": 10.0, "state_mode": "basic", "fixed_green": 25},
    "haichau_peak":  {"count_norm": 40.0, "state_mode": "basic", "fixed_green": 25},
    "haichau":       {"count_norm": 40.0, "state_mode": "basic", "fixed_green": 25},
}


def load_bundle(area, ckpt, count_norm, state_mode):
    rn = RoadNet.load(str(ROOT / "data" / "danang" / "roadnet_{}.json".format(area)))
    enc = ObservationEncoder(rn, count_norm=count_norm, state_mode=state_mode)
    npz = np.load(str(ROOT / ckpt) if not Path(ckpt).is_absolute() else ckpt)
    hidden = npz["W1"].shape[1]
    net = SharedDQN(enc.obs_dim, enc.max_green, hidden=hidden)
    net.load(str(ROOT / ckpt) if not Path(ckpt).is_absolute() else ckpt)
    weights = {k: npz[k].astype(np.float32).tolist() for k in npz.files}
    return rn, enc, net, weights


def forward_mirror(weights, obs):
    """NumPy re-impl of the JS forward pass; source of truth for the parity test."""
    x = np.asarray(obs, dtype=np.float32)
    a1 = np.maximum(x.dot(np.asarray(weights["W1"], np.float32)) + np.asarray(weights["b1"], np.float32), 0.0)
    a2 = np.maximum(a1.dot(np.asarray(weights["W2"], np.float32)) + np.asarray(weights["b2"], np.float32), 0.0)
    if "Wv" in weights:  # dueling
        v = a2.dot(np.asarray(weights["Wv"], np.float32)) + np.asarray(weights["bv"], np.float32)
        adv = a2.dot(np.asarray(weights["Wa"], np.float32)) + np.asarray(weights["ba"], np.float32)
        return v + (adv - adv.mean())
    return a2.dot(np.asarray(weights["W3"], np.float32)) + np.asarray(weights["b3"], np.float32)


def parity_samples(net, obs_dim, n_actions, n, seed):
    rng = np.random.RandomState(seed)
    out = []
    full_mask = np.ones(n_actions, dtype=bool)
    for _ in range(n):
        obs = (rng.rand(obs_dim) * 3.0).astype(np.float32)
        q = net.q_values(obs)[0]
        out.append({"obs": obs.tolist(), "q": q.tolist(),
                    "greedy": int(net.greedy_action(obs, full_mask))})
    return out


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--area", default="haichau_rush")
    ap.add_argument("--ckpt", default=None)
    ap.add_argument("--inter", default=None)
    ap.add_argument("--out", default="frontend/sandbox_model.json")
    ap.add_argument("--samples", type=int, default=32)
    ap.add_argument("--list", action="store_true")
    args = ap.parse_args()

    scen = SCENARIOS.get(args.area, {"count_norm": 5.0, "state_mode": "basic", "fixed_green": 25})
    ckpt = args.ckpt or "results/{a}/dqn_{a}.npz".format(a=args.area)
    rn, enc, net, weights = load_bundle(args.area, ckpt, scen["count_norm"], scen["state_mode"])

    # (Task 2 fills in intersection selection, geometry, phase movements, and the
    # full bundle. Task 1 stops after weights/dims/parity so the parity path is
    # testable on its own.)
    bundle = {
        "meta": {"area": args.area, "state_mode": scen["state_mode"],
                 "count_norm": scen["count_norm"], "fixed_green": scen["fixed_green"],
                 "action_interval": 10, "min_green": 10, "yellow_time": 3,
                 "dueling": "Wv" in weights},
        "dims": {"max_in": enc.max_in, "max_out": enc.max_out, "max_green": enc.max_green,
                 "obs_dim": enc.obs_dim, "hidden": int(np.asarray(weights["W1"]).shape[1]),
                 "n_actions": enc.max_green},
        "weights": weights,
        "parity": parity_samples(net, enc.obs_dim, enc.max_green, args.samples, seed=1),
    }
    out_path = ROOT / args.out
    out_path.write_text(json.dumps(bundle))
    print("wrote", out_path, "obs_dim=", enc.obs_dim, "n_actions=", enc.max_green)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
_COMPASS_VI = ["Đông", "Đông Bắc", "Bắc", "Tây Bắc", "Tây", "Tây Nam", "Nam", "Đông Nam"]


def bearing_deg(cx, cy, px, py):
    return (math.degrees(math.atan2(py - cy, px - cx)) + 360.0) % 360.0


def compass_vi(bearing):
    return _COMPASS_VI[int((bearing + 22.5) // 45) % 8]


def _incoming_road_ids(rn, inter_id):
    return [r for r in rn._by_id[inter_id]["roads"]
            if rn._road_by_id[r]["endIntersection"] == inter_id]


def score_intersection(rn, inter_id):
    greens = rn.green_phases(inter_id)
    lp = rn._lightphases(inter_id)
    n_links = len(rn._by_id[inter_id].get("roadLinks", []))
    # a "clean" green phase serves a strict subset of roadLinks (not all-green)
    clean = sum(1 for g in greens if 0 < len(lp[g].get("availableRoadLinks", [])) < n_links)
    n_in = len(_incoming_road_ids(rn, inter_id))
    return (min(n_in, 4), clean, len(greens))


def pick_intersection(rn):
    return max(rn.signalized_ids(), key=lambda i: score_intersection(rn, i))


def _far_point(rn, road_id, cx, cy):
    pts = rn._road_by_id[road_id]["points"]
    return max(pts, key=lambda p: (p["x"] - cx) ** 2 + (p["y"] - cy) ** 2)


def intersection_block(rn, enc, inter_id):
    inter = rn._by_id[inter_id]
    cx, cy = inter["point"]["x"], inter["point"]["y"]
    incoming = rn.incoming_lanes(inter_id)
    outgoing = rn.outgoing_lanes(inter_id)
    road_links = inter["roadLinks"]
    greens = rn.green_phases(inter_id)
    lp = rn._lightphases(inter_id)

    def road_lane_ids(road_id):
        n = len(rn._road_by_id[road_id]["lanes"])
        return ["{}_{}".format(road_id, k) for k in range(n)]

    def approach_label(road_id):
        fp = _far_point(rn, road_id, cx, cy)
        return compass_vi(bearing_deg(cx, cy, fp["x"], fp["y"]))

    phases = []
    for g in greens:
        movements, appset = [], []
        for rl_idx in lp[g].get("availableRoadLinks", []):
            rl = road_links[rl_idx]
            in_l, out_l = [], []
            for ll in rl["laneLinks"]:
                il = "{}_{}".format(rl["startRoad"], ll["startLaneIndex"])
                ol = "{}_{}".format(rl["endRoad"], ll["endLaneIndex"])
                if il not in in_l:
                    in_l.append(il)
                if ol not in out_l:
                    out_l.append(ol)
            movements.append({"from_road": rl["startRoad"], "to_road": rl["endRoad"],
                              "in_lanes": in_l, "out_lanes": out_l,
                              "turn": rl.get("type", "go_straight")})
            lab = approach_label(rl["startRoad"])
            if lab not in appset:
                appset.append(lab)
        phases.append({"phase_index": g, "movements": movements, "approaches_vi": appset})

    approaches = []
    for road_id in inter["roads"]:
        road = rn._road_by_id[road_id]
        role = "in" if road["endIntersection"] == inter_id else "out"
        fp = _far_point(rn, road_id, cx, cy)
        approaches.append({"road": road_id, "role": role,
                           "angle_deg": bearing_deg(cx, cy, fp["x"], fp["y"]),
                           "compass_vi": compass_vi(bearing_deg(cx, cy, fp["x"], fp["y"])),
                           "lanes": road_lane_ids(road_id)})

    return {"incoming_lanes": incoming, "outgoing_lanes": outgoing,
            "num_greens": len(greens), "green_phases": phases,
            "geometry": {"center": [cx, cy], "approaches": approaches}}
```

```js
// Pure logic for the sandbox demo: portable Net + Encoder (basic state).
// Loads in the browser (window.SandboxCore) and in Node (module.exports) so the
// exact same code path is unit-tested against the trained net's parity samples.
(function (root, factory) {
  const api = factory();
  if (typeof module !== "undefined" && module.exports) module.exports = api;
  else root.SandboxCore = api;
})(typeof self !== "undefined" ? self : this, function () {
  "use strict";

  function linear(x, W, b) {          // x:[in], W:[in][out], b:[out] -> [out]
    const out = b.slice();
    for (let i = 0; i < x.length; i++) {
      const xi = x[i];
      if (xi === 0) continue;
      const Wi = W[i];
      for (let j = 0; j < out.length; j++) out[j] += xi * Wi[j];
    }
    return out;
  }
  function reluInPlace(a) { for (let j = 0; j < a.length; j++) if (a[j] < 0) a[j] = 0; return a; }

  class Net {
    constructor(weights, dims) { this.w = weights; this.dims = dims; }
    forward(x) {
      const w = this.w;
      const a1 = reluInPlace(linear(x, w.W1, w.b1));
      const a2 = reluInPlace(linear(a1, w.W2, w.b2));
      let q;
      if (w.Wv) {                     // dueling recombination
        const v = linear(a2, w.Wv, w.bv);
        const adv = linear(a2, w.Wa, w.ba);
        let mean = 0; for (let j = 0; j < adv.length; j++) mean += adv[j]; mean /= adv.length;
        q = adv.map((a) => v[0] + (a - mean));
      } else {
        q = linear(a2, w.W3, w.b3);
      }
      return Float64Array.from(q);
    }
    maskedArgmax(q, numValid) {
      let best = -1, bestV = -Infinity;
      for (let j = 0; j < numValid; j++) if (q[j] > bestV) { bestV = q[j]; best = j; }
      return best;
    }
  }

  class Encoder {
    constructor(dims, intersection) {
      this.maxIn = dims.max_in; this.maxOut = dims.max_out;
      this.obsDim = dims.obs_dim; this.numGreens = intersection.num_greens;
      this.incoming = intersection.incoming_lanes;
      this.outgoing = intersection.outgoing_lanes;
      this.countNorm = null;          // set by caller from meta.count_norm
    }
    encode(inCounts, outCounts, greenIdx) {
      const v = new Float64Array(this.obsDim);
      const cn = this.countNorm;
      for (let k = 0; k < this.incoming.length; k++) v[k] = (inCounts[this.incoming[k]] || 0) / cn;
      for (let k = 0; k < this.outgoing.length; k++) v[this.maxIn + k] = (outCounts[this.outgoing[k]] || 0) / cn;
      if (greenIdx >= 0 && greenIdx < this.numGreens) v[this.maxIn + this.maxOut + greenIdx] = 1.0;
      return v;
    }
  }

  return { Net, Encoder };
});
```

```js
  function mulberry32(seed) {
    let a = seed >>> 0;
    return function () {
      a |= 0; a = (a + 0x6D2B79F5) | 0;
      let t = Math.imul(a ^ (a >>> 15), 1 | a);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  const SAT_PER_LANE = 0.5;      // veh/s discharged per lane on green
  const OUT_DECAY = 0.3;         // fraction/s outgoing occupancy drains downstream

  class MicroModel {
    constructor(intersection, meta, rng) {
      this.it = intersection; this.meta = meta; this.rng = rng;
      this.t = 0; this.greenIdx = 0; this.sinceSwitch = 0; this.yellowLeft = 0;
      this.inCounts = {}; this.outCounts = {};
      for (const l of intersection.incoming_lanes) this.inCounts[l] = 0;
      for (const l of intersection.outgoing_lanes) this.outCounts[l] = 0;
      // road -> its incoming lane ids (for spreading arrivals)
      this._inLanesByRoad = {};
      for (const ap of intersection.geometry.approaches)
        if (ap.role === "in") this._inLanesByRoad[ap.road] = ap.lanes;
    }
    arrivals(dt, ratePerApproach) {
      for (const road in this._inLanesByRoad) {
        const lanes = this._inLanesByRoad[road];
        const lam = (ratePerApproach[road] || 0) * dt;
        // expected lam arrivals spread across the road's lanes; Bernoulli per unit
        let n = 0; for (let k = 0; k < Math.ceil(lam) + 2; k++) if (this.rng() < lam / (Math.ceil(lam) + 2)) n++;
        for (let i = 0; i < n; i++) this.inCounts[lanes[i % lanes.length]] += 1;
      }
    }
    dischargeAndDecay(dt) {
      if (this.yellowLeft > 0) { this.yellowLeft = Math.max(0, this.yellowLeft - dt); }
      else {
        const ph = this.it.green_phases[this.greenIdx];
        for (const mv of ph.movements) {
          for (const il of mv.in_lanes) {
            const move = Math.min(this.inCounts[il], SAT_PER_LANE * dt);
            this.inCounts[il] -= move;
            const ol = mv.out_lanes[0];
            if (ol != null) this.outCounts[ol] += move;
          }
        }
      }
      for (const ol in this.outCounts) this.outCounts[ol] = Math.max(0, this.outCounts[ol] * (1 - OUT_DECAY * dt));
      this.t += dt; this.sinceSwitch += dt;
    }
    laneCountsIn() { return this.inCounts; }
    laneCountsOut() { return this.outCounts; }
    totalQueue() { let s = 0; for (const l in this.inCounts) s += this.inCounts[l]; return s; }
    applyChoice(choice) {
      const old = this.greenIdx;
      if (choice === old || choice < 0 || choice >= this.it.num_greens
          || this.sinceSwitch < this.meta.min_green) {
        return { changed: false, oldPhase: old, newPhase: old, yellow: 0 };
      }
      this.greenIdx = choice; this.sinceSwitch = 0; this.yellowLeft = this.meta.yellow_time;
      return { changed: true, oldPhase: old, newPhase: choice, yellow: this.meta.yellow_time };
    }
  }
```

```js
  class Agent {
    constructor(model) {
      this.net = new Net(model.weights, model.dims);
      this.enc = new Encoder(model.dims, model.intersection);
      this.enc.countNorm = model.meta.count_norm;
      this.numGreens = model.intersection.num_greens;
      this.nActions = model.dims.n_actions;
    }
    decide(micro) {
      const x = this.enc.encode(micro.laneCountsIn(), micro.laneCountsOut(), micro.greenIdx);
      const q = this.net.forward(x);
      return { choice: this.net.maskedArgmax(q, this.numGreens), q };
    }
  }

  function _phaseQueue(micro, phaseIdx) {   // summed incoming queue on a phase's served lanes
    const ph = micro.it.green_phases[phaseIdx];
    let s = 0;
    for (const mv of ph.movements) for (const il of mv.in_lanes) s += micro.inCounts[il] || 0;
    return s;
  }

  const CONTROLLERS = {
    dqn(model) {
      const ag = new Agent(model);
      return { name: "DQN", labelVi: "RL (DQN)", isRL: true,
               decide(micro) { return ag.decide(micro); } };
    },
    fixed_time(model) {
      const period = model.meta.fixed_green;
      return { name: "FixedTime", labelVi: "Cố định (Fixed-Time)", isRL: false,
               decide(micro, ctx) {
                 const idx = Math.floor((ctx.tSinceCycle || micro.t) / period) % micro.it.num_greens;
                 return { choice: idx, q: null };
               } };
    },
    max_pressure(model) {
      return { name: "MaxPressure", labelVi: "Max-Pressure", isRL: false,
               decide(micro) {
                 let best = 0, bv = -Infinity;
                 for (let g = 0; g < micro.it.num_greens; g++) {
                   const v = _phaseQueue(micro, g);
                   if (v > bv) { bv = v; best = g; }
                 }
                 return { choice: best, q: null };
               } };
    },
    random(model) {
      return { name: "Random", labelVi: "Ngẫu nhiên", isRL: false,
               decide(micro, ctx) {
                 const r = (ctx && ctx.rng) ? ctx.rng() : 0.5;
                 return { choice: Math.floor(r * micro.it.num_greens), q: null };
               } };
    },
  };

  function formatDecision(evt) {
    const head = "[t=" + Math.round(evt.t) + "s] " + evt.ctrlLabel;
    const appr = evt.approachesVi && evt.approachesVi.length ? " · xanh cho hướng " + evt.approachesVi.join("–") : "";
    const why = evt.reason ? " · (" + evt.reason + ")" : "";
    const qs = (evt.qWin != null) ? " · Q=" + evt.qWin.toFixed(1) : "";
    if (evt.changed) return head + " — đổi pha " + evt.oldPhase + "→" + evt.newPhase + appr + why + qs;
    return head + " — giữ pha " + evt.oldPhase + " (đang xanh)" + qs;
  }

  class Scoreboard {
    constructor() { this.acc = {}; }
    accumulate(label, queue, dt) { this.acc[label] = (this.acc[label] || 0) + queue * dt / 3600.0; }
    summary() {
      const fixed = this.acc["Fixed-Time"] || 0;
      const visited = Object.assign({}, this.acc);
      // any non-fixed visible controller vs the fixed shadow
      let cur = 0, curKey = null;
      for (const k in this.acc) if (k !== "Fixed-Time") { cur = this.acc[k]; curKey = k; }
      const deltaPctVsFixed = fixed ? (100 * (fixed - cur) / fixed) : 0;
      return { visited, curKey, deltaPctVsFixed };
    }
  }
```

```js
(function () {
  "use strict";
  const C = window.SandboxCore;

  fetch("sandbox_model.json").then((r) => r.json()).then((model) => start(model));

  function start(model) {
    const canvas = document.getElementById("canvas");
    const ctx = canvas.getContext("2d");
    const W = canvas.width, H = canvas.height;
    const it = model.intersection;

    // world -> screen transform (fit the approaches' far points into the canvas)
    const cx = it.geometry.center[0], cy = it.geometry.center[1];
    let maxR = 1;
    for (const ap of it.geometry.approaches) maxR = Math.max(maxR, 120);
    const scale = (Math.min(W, H) * 0.42) / maxR;
    function sx(dx) { return W / 2 + dx * scale; }
    function sy(dy) { return H / 2 - dy * scale; }   // y up

    const micro = new C.MicroModel(it, model.meta, C.mulberry32(42));
    const state = { model, micro, ctx, W, H, sx, sy, greenIdx: 0, flash: 0 };
    window.__sandbox = state;    // Task 7 attaches controllers/loop; expose for wiring

    drawFrame(state);
  }

  function approachDir(ap) {   // unit vector from center toward the approach's far side
    const a = ap.angle_deg * Math.PI / 180;
    return { x: Math.cos(a), y: Math.sin(a) };
  }

  function drawFrame(s) {
    const { ctx, W, H, sx, sy, model } = s;
    const it = model.intersection;
    ctx.clearRect(0, 0, W, H);
    // roads as thick spokes
    for (const ap of it.geometry.approaches) {
      const d = approachDir(ap);
      ctx.strokeStyle = "#88a"; ctx.lineWidth = 22;
      ctx.beginPath(); ctx.moveTo(sx(0), sy(0)); ctx.lineTo(sx(d.x * 120), sy(d.y * 120)); ctx.stroke();
    }
    // hub
    ctx.fillStyle = s.flash > 0 ? "#fff8b0" : "#c8d4dc";
    ctx.beginPath(); ctx.arc(sx(0), sy(0), 26, 0, Math.PI * 2); ctx.fill();
    // queues + signal heads per incoming approach
    const greenLanes = servedInLanes(it, s.greenIdx);
    for (const ap of it.geometry.approaches) {
      if (ap.role !== "in") continue;
      const d = approachDir(ap);
      const anyGreen = ap.lanes.some((l) => greenLanes.has(l));
      // signal head near the hub
      ctx.fillStyle = anyGreen ? "#85ee00" : "#db635e";
      ctx.beginPath(); ctx.arc(sx(d.x * 42), sy(d.y * 42), 8, 0, Math.PI * 2); ctx.fill();
      // queued cars as stacked ticks going outward
      let q = 0; for (const l of ap.lanes) q += s.micro.inCounts[l] || 0;
      ctx.fillStyle = "#37506a";
      for (let i = 0; i < Math.min(Math.round(q), 20); i++) {
        const rr = 58 + i * 7;
        ctx.fillRect(sx(d.x * rr) - 3, sy(d.y * rr) - 3, 6, 6);
      }
      // count label
      ctx.fillStyle = "#0b1118"; ctx.font = "13px sans-serif";
      ctx.fillText(String(Math.round(q)), sx(d.x * 150) - 6, sy(d.y * 150));
    }
    if (s.flash > 0) s.flash -= 1;
  }

  function servedInLanes(it, greenIdx) {
    const set = new Set();
    const ph = it.green_phases[greenIdx];
    for (const mv of ph.movements) for (const il of mv.in_lanes) set.add(il);
    return set;
  }

  window.__sandboxDraw = drawFrame;         // Task 7 calls this each tick
  window.__sandboxServed = servedInLanes;
```

```js
  function buildControls(s, onChange) {
    const it = s.model.intersection;
    const el = document.getElementById("controls");
    const ctrlOpts = [["dqn", "RL (DQN)"], ["fixed_time", "Cố định"], ["max_pressure", "Max-Pressure"], ["random", "Ngẫu nhiên"]]

    document.getElementById("reset").onclick = () => onChange.reset();
  }

  function approachesViForPhase(it, greenIdx) {
    return it.green_phases[greenIdx].approaches_vi || [];
  }

  function runLoop(s) {
    const C = window.SandboxCore, it = s.model.intersection, meta = s.model.meta;
    let ctrlKey = "dqn", demand = 0.35, speed = 6, playing = true;
    let visible = s.micro;
    let shadow = new C.MicroModel(it, meta, C.mulberry32(42));   // same seed => same arrivals
    let visCtrl = C.CONTROLLERS[ctrlKey](s.model);
    let fixedCtrl = C.CONTROLLERS.fixed_time(s.model);
    const rng = C.mulberry32(99);
    const sb = new C.Scoreboard();
    let sinceDecision = 0;
    const rate = () => {
      const r = {}; for (const ap of it.geometry.approaches) if (ap.role === "in") r[ap.road] = demand; return r;
    };

    function decide(micro, ctrl, label, isRL) {
      const { choice, q } = ctrl.decide(micro, { tSinceCycle: micro.t, rng });
      const res = micro.applyChoice(choice);
      s.greenIdx = visible.greenIdx;
      if (micro === visible) {
        s.flash = 6;
        updateQBars(s, q);
        updatePhaseBanner(s);
        const appr = approachesViForPhase(it, micro.greenIdx);
        let reason = "";
        if (q) { let mx = -Infinity, mi = 0; for (let j = 0; j < it.num_greens; j++) if (q[j] > mx) { mx = q[j]; mi = j; } reason = "Q lớn nhất ở pha " + mi; }
        logLine(s, C.formatDecision({ t: micro.t, ctrlLabel: label, changed: res.changed,
          oldPhase: res.oldPhase, newPhase: res.newPhase, approachesVi: appr, reason,
          qWin: q ? Math.max.apply(null, Array.from(q).slice(0, it.num_greens)) : null }), isRL);
      }
    }

    function tick() {
      if (playing) {
        const dt = 1.0;
        for (let sub = 0; sub < speed; sub++) {
          visible.arrivals(dt, rate()); shadow.arrivals(dt, rate());
          visible.dischargeAndDecay(dt); shadow.dischargeAndDecay(dt);
          sb.accumulate(visCtrl.labelVi === "Cố định (Fixed-Time)" ? "Fixed-Time" : visCtrl.name === "DQN" ? "DQN" : visCtrl.name, visible.totalQueue(), dt);
          sb.accumulate("Fixed-Time", shadow.totalQueue(), dt);
          sinceDecision += dt;
          if (sinceDecision >= meta.action_interval) {
            sinceDecision = 0;
            decide(visible, visCtrl, visCtrl.labelVi, visCtrl.isRL);
            decide(shadow, fixedCtrl, "Fixed-Time(bóng)", false);
          }
        }
        updateScoreboard(s, sb);
      }
      window.__sandboxDraw(s);
      requestAnimationFrame(tick);
    }

    buildControls(s, {
      ctrl: (v) => { ctrlKey = v; visCtrl = C.CONTROLLERS[v](s.model); },
      demand: (v) => { demand = v; },
      speed: (v) => { speed = v; },
      togglePlay: () => { playing = !playing; document.getElementById("play").textContent = playing ? "Tạm dừng" : "Tiếp tục"; },
      reset: () => {
        visible = s.micro = new C.MicroModel(it, meta, C.mulberry32(42));
        shadow = new C.MicroModel(it, meta, C.mulberry32(42));
        sb.acc = {}; document.getElementById("log").innerHTML = ""; s.greenIdx = 0;
      },
    });
    tick();
  }

  function updatePhaseBanner(s) {
    const it = s.model.intersection;
    document.getElementById("phase-label").textContent =
      "#" + s.greenIdx + " (" + (it.green_phases[s.greenIdx].approaches_vi || []).join("–") + ")";
  }
  function updateQBars(s, q) {
    const it = s.model.intersection, host = document.getElementById("qbars");
    if (!q) { host.innerHTML = "<div class='note'>Baseline không dùng Q-value.</div>"; return; }
    let mx = -Infinity, mi = 0; for (let j = 0; j < it.num_greens; j++) if (q[j] > mx) { mx = q[j]; mi = j; }
    let mn = Infinity; for (let j = 0; j < it.num_greens; j++) mn = Math.min(mn, q[j]);
    let html = "<div class='note'>Q-value mỗi pha (DQN):</div>";
    for (let j = 0; j < it.num_greens; j++) {
      const w = Math.round(100 * (q[j] - mn) / (mx - mn + 1e-6));
      html += `<div class="qbar ${j === mi ? "win" : ""}"><i style="width:${w}%"></i><span>pha ${j}: ${q[j].toFixed(2)}</span></div>`;
    }
    host.innerHTML = html;
  }
  function updateScoreboard(s, sb) {
    const sum = sb.summary(), host = document.getElementById("scoreboard");
    const cur = sum.curKey ? sum.visited[sum.curKey] : 0, fixed = sum.visited["Fixed-Time"] || 0;
    const arrow = sum.deltaPctVsFixed >= 0 ? "▼" : "▲";
    host.innerHTML = `<div><b>Thời gian chờ tích luỹ (trong sandbox)</b></div>` +
      `<div>${sum.curKey || "—"}: ${cur.toFixed(2)} veh-h</div>` +
      `<div>Fixed-Time (bóng): ${fixed.toFixed(2)} veh-h</div>` +
      `<div>${arrow} ${Math.abs(sum.deltaPctVsFixed).toFixed(1)}% so với Fixed-Time</div>`;
  }
  function logLine(s, text, isRL) {
    const host = document.getElementById("log");
    const div = document.createElement("div");
    div.className = "log-line " + (isRL ? "rl" : "base");
    div.textContent = text;
    host.insertBefore(div, host.firstChild);
    while (host.childNodes.length > 200) host.removeChild(host.lastChild);
  }

  // kick off once the model+state are ready (start() sets window.__sandbox)
  const _boot = setInterval(() => {
    if (window.__sandbox) { clearInterval(_boot); runLoop(window.__sandbox); }
  }, 30);
})();
```
